/* ═════════ ENGINE — thay bằng thuật toán thật của bạn ═════════
   Trả về { pick:'P'|'B'|null, conf, pP, pB, pT, good, tag, why, log:[] }
   pP + pB + pT = 100.  conf = độ nghiêng giữa Con và Cái.                */
function predict(hist){
  const n = hist.length, seq = hist.filter(x => x !== 'T');
  const tieN = hist.length - seq.length;
  const tieP = Math.max(5, Math.min(14, Math.round(9.5*.6 + (n ? tieN/n*100 : 9.5)*.4)));
  const log = [`tai_ban  so_van=${n}`, `loc_hoa   con_lai=${seq.length}`];
  if (seq.length < 3){
    const pP = Math.round((100-tieP)*.49), pB = 100-tieP-pP;
    log.push('canh_bao  chua_du_du_lieu (<3 van)');
    return { pick:null, conf:0, pP, pB, pT:tieP, good:false, tag:'', why:'Cần ít nhất 3 ván (không tính Hòa).', log };
  }
  const p = seq.filter(x => x==='P').length, b = seq.length - p;
  let run = 1; for (let i = seq.length-1; i > 0 && seq[i] === seq[i-1]; i--) run++;
  const last = seq.at(-1), other = last === 'P' ? 'B' : 'P';
  const tail = seq.slice(-4), alt = tail.length >= 4 && tail.every((v,i) => i === 0 || v !== tail[i-1]);

  let P = 44.6, B = 45.9, good = false, pat, why, tag = '';
  const add = (k, v) => { k === 'P' ? (P += v) : (B += v); };
  if (run >= 3){ add(last, Math.min(3*run, 16)); good = run >= 4; tag = `Bệt ${run} ${NAME[last]}`; pat = `cau_bet  ${run}x${NAME[last]}`; why = `Chuỗi ${run} ván ${NAME[last]} liên tiếp, tín hiệu theo cầu bệt.`; }
  else if (alt){ add(other, 9); good = true; tag = 'Nhảy 1-1'; pat = 'cau_nhay  1-1'; why = 'Bốn ván gần nhất đổi bên liên tục, tín hiệu theo cầu nhảy.'; }
  else { pat = 'cau_chua_ro_hinh'; why = 'Cầu chưa rõ hình, nghiêng nhẹ về bên đang nhiều ván hơn.'; }
  const bias = (p - b) / seq.length * 6; P += bias; B -= bias;

  const s = P + B; P = P/s*(100-tieP); B = B/s*(100-tieP);
  const pP = Math.round(P), pB = 100 - tieP - pP;
  const pick = pB >= pP ? 'B' : 'P';
  const conf = Math.round(Math.max(pP,pB) / (pP+pB) * 100);
  log.push(pat, `lech_cau  P=${p} B=${b}`, `mo_hinh   P ${pP}%  B ${pB}%  T ${tieP}%`);
  return { pick, conf, pP, pB, pT:tieP, good:good || conf >= 58, tag, why, log };
}

    if (t.state !== 'shuffle' && r.pick) el.style.setProperty('--a', COLOR[r.pick]); else el.style.removeProperty('--a');
    if (fresh){ void el.offsetWidth; el.classList.add('fresh'); }
    return el;
  });
  if (!els.length) list.innerHTML = '<div class="none">Chưa có bàn nào có cầu đẹp. Tín hiệu sẽ hiện khi có chuỗi rõ.</div>';
  else if (list.children.length !== els.length || els.some((e,i) => list.children[i] !== e)) list.replaceChildren(...els);
}

function attach3d(stage, holo){
  const o = { touching:false, set(ry, rx){ holo.style.setProperty('--ry', ry + 'deg'); holo.style.setProperty('--rx', rx + 'deg'); } };
  const move = e => {
    const r = stage.getBoundingClientRect();
    const x = ((e.clientX - r.left) / r.width - .5) * 2, y = ((e.clientY - r.top) / r.height - .5) * 2;
    o.set(Math.max(-1,Math.min(1,x)) * 34, -Math.max(-1,Math.min(1,y)) * 28);
  };
  stage.addEventListener('pointerdown', e => { o.touching = true; stage.style.cursor = 'grabbing'; move(e); buzz(6); });
  stage.addEventListener('pointermove', e => { if (o.touching || e.pointerType === 'mouse') move(e); });
  ['pointerup','pointercancel','pointerleave'].forEach(ev => stage.addEventListener(ev, () => { o.touching = false; stage.style.cursor = ''; o.set(0, 0); }));
  return o;
}

}
